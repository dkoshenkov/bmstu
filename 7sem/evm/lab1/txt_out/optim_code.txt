        .section .text
        .globl _start;
        len = 8             # Размер массива
        enroll = 4          # Количество обрабатываемых элементов за одну итерацию
        elem_sz = 4         # Размер одного элемента массива

_start:
        la      x1, _x
        addi    x20, x1, elem_sz*len   # Адрес последнего элемента
        add     x31, x0, x0
lp:
    lw      x2, 0(x1)        # 1. Начали загрузку x2
    lw      x3, 4(x1)        # 2. Пока x2 грузится, начали x3
    lw      x4, 8(x1)        # 3. Пока предыдущие грузятся, начали x4
    lw      x5, 12(x1)       # 4. И x5 тоже
    add     x31, x31, x2     # 5. Теперь x2 УЖЕ готов! Складываем без ожидания
    add     x31, x31, x3     # 6. x3 тоже готов
    add     x31, x31, x4
    add     x31, x31, x5
    addi    x1, x1, 16
    bne     x1, x20, lp

lp2:
        j       lp2

        .section .data
_x:
        .4byte  0x1
        .4byte  0x2
        .4byte  0x3
        .4byte  0x4
        .4byte  0x5
        .4byte  0x6
        .4byte  0x7
        .4byte  0x8
